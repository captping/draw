from pathlib import Path
import xml.etree.ElementTree as ET

OUTPUT_FILE = Path.home() / "Desktop" / "mobile_security_flow.drawio"

mxfile = ET.Element("mxfile", {"host": "app.diagrams.net"})
diagram = ET.SubElement(mxfile, "diagram", {"name": "Mobile Security Flow"})
model = ET.SubElement(diagram, "mxGraphModel", {
    "dx": "1500", "dy": "900", "grid": "0", "gridSize": "10",
    "guides": "1", "tooltips": "1", "connect": "1", "arrows": "1",
    "fold": "1", "page": "1", "pageScale": "1",
    "pageWidth": "1500", "pageHeight": "900", "math": "0", "shadow": "0"
})

root = ET.SubElement(model, "root")
ET.SubElement(root, "mxCell", {"id": "0"})
ET.SubElement(root, "mxCell", {"id": "1", "parent": "0"})


def add_box(id, text, x, y, w, h, fill="#ffffff", stroke="#2f5597", size="15", bold=False):
    style = (
        f"rounded=1;whiteSpace=wrap;html=1;fillColor={fill};strokeColor={stroke};"
        f"fontColor=#1f2937;fontSize={size};spacing=10;strokeWidth=2;"
    )
    if bold:
        style += "fontStyle=1;"
    cell = ET.SubElement(root, "mxCell", {
        "id": id, "value": text, "style": style, "vertex": "1", "parent": "1"
    })
    ET.SubElement(cell, "mxGeometry", {
        "x": str(x), "y": str(y), "width": str(w), "height": str(h), "as": "geometry"
    })


def add_cloud(id, text, x, y, w, h):
    cell = ET.SubElement(root, "mxCell", {
        "id": id,
        "value": text,
        "style": "shape=cloud;whiteSpace=wrap;html=1;fillColor=#ffffff;strokeColor=#2f5597;fontSize=18;fontStyle=1;strokeWidth=2;",
        "vertex": "1",
        "parent": "1"
    })
    ET.SubElement(cell, "mxGeometry", {
        "x": str(x), "y": str(y), "width": str(w), "height": str(h), "as": "geometry"
    })


def add_edge(id, source, target, label="", dashed=False):
    style = (
        "edgeStyle=orthogonalEdgeStyle;rounded=0;orthogonalLoop=1;jettySize=auto;"
        "html=1;endArrow=block;endFill=1;strokeWidth=2;strokeColor=#6b7280;fontSize=12;"
    )
    if dashed:
        style += "dashed=1;"
    cell = ET.SubElement(root, "mxCell", {
        "id": id, "value": label, "style": style, "edge": "1",
        "parent": "1", "source": source, "target": target
    })
    ET.SubElement(cell, "mxGeometry", {"relative": "1", "as": "geometry"})
